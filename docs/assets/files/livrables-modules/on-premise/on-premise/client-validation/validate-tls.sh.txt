#!/bin/sh
set -eu

check_tls() {
    label=$1
    host=$2
    port=$3
    starttls=${4:-}
    output=$(mktemp)

    printf '\n== %s: %s:%s ==\n' "$label" "$host" "$port"
    if [ -n "$starttls" ]; then
        openssl s_client -starttls "$starttls" -connect "$host:$port" \
            -servername "$host" -verify_hostname "$host" \
            -verify_return_error </dev/null >"$output" 2>&1
    else
        openssl s_client -connect "$host:$port" -servername "$host" \
            -verify_hostname "$host" -verify_return_error \
            </dev/null >"$output" 2>&1
    fi

    sed -n '/-----BEGIN CERTIFICATE-----/,/-----END CERTIFICATE-----/p' "$output" \
        | openssl x509 -noout -subject -issuer -dates -ext subjectAltName
    grep 'Verify return code:' "$output"
    grep -q 'Verify return code: 0 (ok)' "$output"
    rm -f "$output"
}

check_tls "OpenLDAP LDAPS" ldap.embedded.local 636
check_tls "Dovecot IMAPS" imap.embedded.local 993
check_tls "Postfix SMTP STARTTLS" smtp.embedded.local 587 smtp
check_tls "Roundcube HTTPS" webmail.embedded.local 8443

printf '\n== Roundcube HTTPS ==\n'
curl --fail --silent --show-error --output /dev/null \
    --write-out 'HTTP %{http_code}\n' https://webmail.embedded.local:8443/

printf '\nTous les certificats sont valides depuis le conteneur client.\n'
